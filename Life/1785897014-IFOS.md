#Usage
Perform state update that triggers side effects

#Syntax
const { useActionState } from 'react'

const [state, dispatchAction, isPending] = useActionState(reducer, initialValue)

#How to invoke

const handler = () => dispatchAction()

- You can handle multiple action via the reducer
const CustomReducer = (currentValue, payload) => {
    switch(payload.type) {
        case: 'ADD': {
            // do something
        }
    }
}
dispatchAction({type: 'ADD})
