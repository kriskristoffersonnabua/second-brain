#Usage
- Perform non blocking state updates

#Syntax
import { useTransition } from 'react'
const [isPending, startTransition] = useTransition()

function TransitionExample() {
  const [isPending, startTransition] = useTransition();
  const [list, setList] = useState([]);
  const [inputValue, setInputValue] = useState("");
  const LIMIT_SIZE = 20000;

  const handleChange = (e) => {
    setInputValue(e.target.value);
    startTransition(() => {
      const tempList = [];
      for (let i = 0; i < LIMIT_SIZE; i++) {
        tempList.push(e.target.value);
      }
      setList(tempList);
    });
  };

  return (
    <div className="App">
      <input type="text" onChange={handleChange} />
      {isPending ? (
        <div>Loading...</div>
      ) : (
        list.map((l, i) => <div key={i}>{l}</div>)
      )}
    </div>
  );
}

#RuleOfThumb
- best use this when you control the state within your compoenent not from other sources like props, url params, 3rd party state
